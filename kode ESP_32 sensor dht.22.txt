#include <WiFi.h>
#include <PubSubClient.h>
#include "DHT.h"


// --- Konfigurasi WiFi & MQTT ---
const char* ssid = "FR-SYSTEM";
const char* password = "bebekgoreng";
const char* mqtt_server = "52.91.170.236"; // Kamu bisa pakai broker publik/pribadi


#define DHTPIN 26        
#define DHTTYPE DHT22    


DHT dht(DHTPIN, DHTTYPE);
WiFiClient espClient;
PubSubClient client(espClient);


unsigned long lastMsg = 0;


void setup_wifi() {
  delay(10);
  Serial.println();
  Serial.print("Menghubungkan ke ");
  Serial.println(ssid);


  WiFi.begin(ssid, password);


  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }


  Serial.println("\nWiFi terhubung!");
}


void reconnect() {
  // Loop sampai terhubung kembali
  while (!client.connected()) {
    Serial.print("Mencoba koneksi MQTT...");
    // ID Client unik
    String clientId = "mqttx_0e1d43c9";
    clientId += String(random(0xffff), HEX);
   
    if (client.connect(clientId.c_str(), "tema123", "123")) {
      Serial.println("terhubung");
    } else {
      Serial.print("gagal, rc=");
      Serial.print(client.state());
      Serial.println(" coba lagi dalam 5 detik");
      delay(5000);
    }
  }
}


void setup() {
  Serial.begin(115200);
  setup_wifi();
  client.setServer(mqtt_server, 1883);
  dht.begin();
}


void loop() {
  if (!client.connected()) {
    reconnect();
  }
  client.loop();


  unsigned long now = millis();
  // Kirim data setiap 5 detik (agar tidak spamming broker)
  if (now - lastMsg > 5000) {
    lastMsg = now;


    float temperature = dht.readTemperature();
    float humidity = dht.readHumidity();


    if (isnan(temperature) || isnan(humidity)) {
      Serial.println("Gagal membaca dari sensor DHT!");
      return;
    }


    // Konversi float ke string untuk dikirim
    char tempString[8];
    dtostrf(temperature, 1, 2, tempString);
    char humString[8];
    dtostrf(humidity, 1, 2, humString);


    // Publish ke Topic
    client.publish("esp32/suhu", tempString);
    client.publish("esp32/kelembapan", humString);


    Serial.print("Suhu Terkirim: ");
    Serial.println(tempString);
    Serial.print("Kelembapan Terkirim: ");
    Serial.println(humString);
  }
}
